#[soroban_sdk::contractargs(name = "Args")]
#[soroban_sdk::contractclient(name = "Client")]
pub trait Contract {
    fn settle(
        env: soroban_sdk::Env,
        owner: soroban_sdk::Address,
        campaign_id: u64,
        code: soroban_sdk::String,
        period: u64,
    ) -> Result<soroban_sdk::Vec<Payout>, soroban_sdk::Error>;
    fn verify(
        env: soroban_sdk::Env,
        campaign_id: u64,
        code: soroban_sdk::String,
    ) -> Result<CouponToken, soroban_sdk::Error>;
    fn is_valid(
        env: soroban_sdk::Env,
        campaign_id: u64,
        code: soroban_sdk::String,
    ) -> bool;
    fn get_tally(
        env: soroban_sdk::Env,
        campaign_id: u64,
        code: soroban_sdk::String,
        period: u64,
    ) -> Result<TallyCommitment, soroban_sdk::Error>;
    fn bump_codes(
        env: soroban_sdk::Env,
        campaign_id: u64,
        codes: soroban_sdk::Vec<soroban_sdk::String>,
    ) -> Result<(), soroban_sdk::Error>;
    fn bump_tally(
        env: soroban_sdk::Env,
        campaign_id: u64,
        code: soroban_sdk::String,
        periods: soroban_sdk::Vec<u64>,
    ) -> Result<(), soroban_sdk::Error>;
    fn get_shared(
        env: soroban_sdk::Env,
        campaign_id: u64,
        code: soroban_sdk::String,
    ) -> Result<SharedCode, soroban_sdk::Error>;
    fn is_settled(
        env: soroban_sdk::Env,
        campaign_id: u64,
        code: soroban_sdk::String,
        period: u64,
    ) -> bool;
    fn is_delegate(
        env: soroban_sdk::Env,
        campaign_id: u64,
        who: soroban_sdk::Address,
    ) -> bool;
    fn add_delegate(
        env: soroban_sdk::Env,
        owner: soroban_sdk::Address,
        campaign_id: u64,
        delegate: soroban_sdk::Address,
    ) -> Result<(), soroban_sdk::Error>;
    fn campaigns_of(
        env: soroban_sdk::Env,
        owner: soroban_sdk::Address,
    ) -> soroban_sdk::Vec<u64>;
    fn commit_tally(
        env: soroban_sdk::Env,
        owner: soroban_sdk::Address,
        campaign_id: u64,
        code: soroban_sdk::String,
        period: u64,
        count: u32,
        merkle_root: soroban_sdk::BytesN<32>,
        per_attribution: soroban_sdk::Map<soroban_sdk::Address, u32>,
    ) -> Result<(), soroban_sdk::Error>;
    fn get_campaign(
        env: soroban_sdk::Env,
        campaign_id: u64,
    ) -> Result<Campaign, soroban_sdk::Error>;
    fn issue_unique(
        env: soroban_sdk::Env,
        owner: soroban_sdk::Address,
        campaign_id: u64,
        codes: soroban_sdk::Vec<soroban_sdk::String>,
    ) -> Result<soroban_sdk::Vec<u64>, soroban_sdk::Error>;
    fn total_minted(env: soroban_sdk::Env) -> u64;
    fn bump_campaign(
        env: soroban_sdk::Env,
        campaign_id: u64,
    ) -> Result<(), soroban_sdk::Error>;
    fn redeem_unique(
        env: soroban_sdk::Env,
        authorizer: soroban_sdk::Address,
        campaign_id: u64,
        code: soroban_sdk::String,
        redeemer_ref_hash: soroban_sdk::BytesN<32>,
    ) -> Result<RedemptionReceipt, soroban_sdk::Error>;
    fn bump_delegates(
        env: soroban_sdk::Env,
        campaign_id: u64,
        delegates: soroban_sdk::Vec<soroban_sdk::Address>,
    ) -> Result<(), soroban_sdk::Error>;
    fn campaign_stats(
        env: soroban_sdk::Env,
        campaign_id: u64,
    ) -> Result<CampaignStats, soroban_sdk::Error>;
    fn campaigns_page(
        env: soroban_sdk::Env,
        owner: soroban_sdk::Address,
        cursor: u64,
        limit: u32,
    ) -> Result<soroban_sdk::Vec<u64>, soroban_sdk::Error>;
    fn compute_payouts(
        env: soroban_sdk::Env,
        campaign_id: u64,
        code: soroban_sdk::String,
        period: u64,
    ) -> Result<soroban_sdk::Vec<Payout>, soroban_sdk::Error>;
    fn create_campaign(
        env: soroban_sdk::Env,
        owner: soroban_sdk::Address,
        name: soroban_sdk::String,
        discount_type: soroban_sdk::String,
        discount_value: u64,
        total_supply: u32,
        valid_until: u64,
    ) -> Result<u64, soroban_sdk::Error>;
    fn register_shared(
        env: soroban_sdk::Env,
        owner: soroban_sdk::Address,
        campaign_id: u64,
        code: soroban_sdk::String,
        attributed_to: Option<soroban_sdk::Address>,
        payout_token: Option<soroban_sdk::Address>,
        payout_rate: i128,
    ) -> Result<(), soroban_sdk::Error>;
    fn remove_delegate(
        env: soroban_sdk::Env,
        owner: soroban_sdk::Address,
        campaign_id: u64,
        delegate: soroban_sdk::Address,
    ) -> Result<(), soroban_sdk::Error>;
    fn total_campaigns(env: soroban_sdk::Env) -> u64;
}
